cmake_minimum_required(VERSION 3.16)
project(sum_coloring_search LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SUM_COLORING_OPENMP "Enable parallel replica sweeps" OFF)
option(SUM_COLORING_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(assignment STATIC third_party/scipy/rectangular_lsap.cpp)
add_executable(sum_coloring src/sum_coloring.cpp)
target_link_libraries(sum_coloring PRIVATE assignment)
target_compile_options(sum_coloring PRIVATE -Wall -Wextra -Wpedantic)
if(SUM_COLORING_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_link_libraries(sum_coloring PRIVATE OpenMP::OpenMP_CXX)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(check_core tests/check_core.cpp)
    target_link_libraries(check_core PRIVATE assignment)
    target_compile_options(check_core PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME core_oracles COMMAND check_core)
    set_tests_properties(core_oracles PROPERTIES TIMEOUT 60)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME command_line COMMAND ${Python3_EXECUTABLE}
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli.py $<TARGET_FILE:sum_coloring>)
    set_tests_properties(command_line PROPERTIES TIMEOUT 60)
endif()
if(SUM_COLORING_SANITIZERS)
    foreach(target assignment sum_coloring check_core)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endforeach()
endif()
